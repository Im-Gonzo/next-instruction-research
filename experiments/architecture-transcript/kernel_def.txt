KERNEL — a shared architecture vocabulary (defined once, reused for every system).
Nodes are `id:type "label"`. Edges: a->b (request/flow), a~>b (async via queue),
a=>b (provisions/creates), a:>b (config flow), a^b (hosts/runs), a&b (references).
Types: client, api, worker, queue, db, store, ctrl, proxy, lb, edge, svc, infra, cfg, sidecar.

PATTERNS (macros that expand to subgraphs):
  async_broker(api,queue,worker,db):
    client->api ; api~>queue ; queue->worker ; worker->db ; api->db
  control_plane(ctrl,proxy,store,broker):   # Envoy xDS plane
    ctrl->broker(pull context) ; store->context
    ctrl:>context ; ctrl:>templates ; context:>templates
    context=>clusters ; templates=>routes ; templates=>listeners
    proxy->clusters ; proxy->routes ; proxy->listeners   (xDS: CDS/RDS/LDS)
  proxy_fleet(cfn,ami,proxy):
    cfn=>{parameters,vpc,igw,sg,keypair,iamrole,asg,nlb,acm,route53} ; vpc=>subnet
    asg=>ec2 ; asg&ami ; ec2^proxy
  image_pipeline(cfgtool,builder)=>ami:
    cfg:>builder ; builder=>ami ; ami&ec2
  edge_chain(client,...hops...,svc; [front]):
    client->front->...->svc (each hop forwards)
  sidecar(proxy,name,lang,owner):
    proxy->name(ext) ; ec2^name
  behind_edge(proxy,[svcs...]):
    proxy->each svc
